     range(num_layers)\n        ])\n        \n        self.layer_norm = LayerNorm(d_model)    # Final normalization\n
     \n    def forward(self, x):\n        \"\"\"Process input through all layers\"\"\"\n        h = x  # Initial hidden
     state\n        \n        for layer_idx in range(len(self.layers)):\n            h = self.layers[layer_idx](h)
     # Sequential processing\n        \n        return self.layer_norm(h)               # Final normalized
     output\n\nclass TransformerDecoder(nn.Module):\n    \"\"\"Transformer decoder with additional masking\"\"\"\n    \n
     def __init__(self, num_layers, d_model, n_heads, feedforward_dim, vocab_size):\n        super().__init__()\n
     \n        self.layers = nn.ModuleList([\n            DecoderBlock(d_model, n_heads) \n            for _ in
     range(num_layers)\n        ])\n        \n        # Output linear layer to vocabulary: maps hidden state → token
     logits\n        self.output_projection = Linear(d_model, vocab_size)\n    \n    def forward(self, x):\n
     \"\"\"Process decoder input through all layers\"\"\"\n        h = x\n        \n        for layer_idx in
     range(len(self.layers)):\n            h = self.layers[layer_idx](h)\n        \n        return
     self.output_projection(h)\n```\n\n---\n\n## 6. Attention Mechanisms Deep Dive\n\n### Self-Attention Mathematical
     Derivation\n\n#### Scaling Dot-Product Attention (Original Paper, Figure 1)\n\n**Step-by-step
     computation:**\n\nGiven:\n- Queries Q ∈ ℝ^{n×d} (batch of n queries, each dimension d)\n- Keys K ∈ ℝ^{m×d} (m
     possible positions to attend from)  \n- Values V ∈ ℝ^{m×e} (values at each position, output dim
     e)\n\n**Computation:**\n\n```python\ndef scaled_dot_product_attention(Q, K, V):\n    \"\"\"Computes attention
     weights and weighted values\n    \n    Q: [batch_size, num_queries, d]\n    K: [batch_size, num_keys, d]  \n    V:
     [batch_size, num_values, e]\n    \n    Returns output of shape [batch_size, num_queries, e]\n    \"\"\"\n    \n
     # 1. Compute attention logits (dot product similarity):\n    # Each query attends to all keys via dot product\n
     # Shape propagation: \n    #   Q^T is [d × n], K is [m × d]  \n    #   Result: A = Q · K^T has shape [n × m] per
     batch element\n    \n    logits = torch.matmul(Q, K.transpose(-2, -1))\n    \n    # 2. Scale by √d to prevent large
     dot products from vanishing softmax gradient\n    scale_factor = d ** (-0.5)\n    scaled_logits = logits *
     scale_factor\n    \n    # 3. Apply softmax (along key dimension m):\n    attn_weights =
     torch.softmax(scaled_logits, dim=-1)  \n    # Each row now sums to 1: represents attention distribution over keys\n
     \n    # 4. Weighted sum of values:\n    output = torch.matmul(attn_weights, V)\n    \n    return output\n\n# Shape
     verification example:\n\"\"\"\nInput shapes for single batch element (no batch dimension shown):\nQ:
     [seq_len_query, d_model]          → queries from current position(s) \nK: [context_len, d_model]            → all
     possible keys in sequence\nV: [context_len, hidden_dim_out]     → values to retrieve\n\nOutput shape:
     [seq_len_query, hidden_dim_out]\n\nExample with BERT-style encoder (no query/key/value distinction):\nQ = K = V = X
     where X is input representation.\nThen attention becomes self-attention connecting all sequence
     positions.\n\"\"\"\n```\n\n#### Multi-Head Attention (Figure 2)\n\n**Parallel computation across multiple
     subspaces:**\n\n```python\ndef multi_head_attention(Q, K, V, num_heads):\n    \"\"\"Split into h parallel attention
     mechanisms\n    \n    Each head learns different types of relationships:\n      - Head 1 might focus on syntactic
     dependencies  \n      - Head 2 might capture semantic similarity\n      - Head 3 might detect long-range
     connections\n      \n    These operate in parallel and are concatenated at the end.\n    \"\"\"\n    \n    d_model
     = Q.shape[-1]  # Feature dimension\n    \n    assert num_heads > 0, \"Must have at least one attention head\"\n
     assert d_model % num_heads == 0, f\"d_model {d_model} must be divisible by heads {num_heads}\"\n    \n    head_dim
     = d_model // num_heads   # Features per head\n    \n    # Split each input along feature dimension:\n    Q_heads =
     torch.cat([Q.split(head_dim, dim=-1) for _ in range(num_heads)], dim=0).transpose(0, -2)\n    K_heads =
     torch.cat([K.split(head_dim, dim=-1) for _ in range(num_heads)], dim=0).transpose(0, -2)  \n    V_heads =
     torch.cat([V.split(head_dim, dim=-1) for _ in range(num_heads)], dim=0).transpose(0, -2)\n    \n    # Apply scaled
     dot-product attention to each head independently:\n    attn_outputs = [scaled_dot_product_attention(Q_h, K_h, V_h)
     \n                    for Q_h, K_h, V_h in zip(*[Q_heads, K_heads, V_heads])]\n    \n    # Concatenate outputs
     along feature dimension and project back:\n    output_concat = torch.cat(attn_outputs, dim=-1)  # [batch ×
     query_len × d_model]\n```\n\n### Positional Encoding (Critical for Transformers!)\n\n**Why needed:** Self-attention
     is permutation-invariant - it doesn't know word order. Must inject positional information separately from
     content.\n\n#### Original Transformer Sinusoidal Encodings\n\n```python\ndef get_sinusoidal_encoding(max_len,
     dim_model):\n    \"\"\"Generate sinusoidal position embeddings (original paper method)\"\"\"\n    \n    # Create
     positions: [0, 1, 2, ..., max_len-1] with shape [max_len]\n    positions = torch.arange(max_len).unsqueeze(1)  #
     Add dimension for broadcasting\n    \n    # Compute frequencies at each even dimension using different scales\n
     div_term = torch.exp(torch.arange(0, dim_model, 2) * (-math.log(10000.0) / (dim_model // 2)))\n    \n    pe =
     torch.zeros(max_len, dim_model)\n    \n    # Even dimensions: cosine waves with decreasing frequency\n    for i in
     range(dim_model):\n        if i % 2 == 0:\n            freq = div_term[i]\n            \n            # Positional
     encoding formula: cos(position * scale^(dimension/2))\n            pe[:, i] = torch.cos(positions * freq)\n    \n
     return pe\n\n# Example usage in transformer model:\n\"\"\"\nclass TransformerModel(nn.Module):\n    def
     __init__(self, vocab_size, d_model, n_heads, num_layers, max_len=512):\n        super().__init__()\n        \n
     self.embedding = nn.Embedding(vocab_size, d_model)  # Token → embedding\n        self.pos_encoding =
     get_sinusoidal_encoding(max_len, d_model)  \n        self.encoder = TransformerEncoder(num_layers, ...)\n    \n
     def forward(self, input_ids):\n        \"\"\"input_ids shape: [batch × seq_len]\"\"\"\n        \n
     batch_size, seq_len = input_ids.shape\n        \n        # Embed tokens to vectors of dimension d_model:\n        x
     = self.embedding(input_ids)  # Shape now: [batch × seq_len × d_model]\n        \n        # Add positional encoding
     (broadcast across positions):\n        pos_enc = self.pos_encoding[:seq_len].unsqueeze(0).expand(batch_size, -1,
     -1)\n        x = x + pos_enc                # Content + position information\n        \n        return
     self.encoder(x)         # Pass through transformer layers\n\"\"\"\n\n# Alternative: Learnable Positional Embeddings
     (used in BERT-style models)\nclass LearnedPositionEmbedding(nn.Module):\n    def __init__(self, max_len,
     d_model):\n        super().__init__()\n        self.embedding = nn.Embedding(max_len, d_model)  # Trainable
     parameters!\n        \n    def forward(self, positions):\n        return self_embedding(positions)\n\n\"\"\"\n# In
     practice: sinusoidal allows OOV length handling; learned is more data-efficient.\n# Modern models like GPT-3 use
     absolute position embeddings (learned).\n\"\"\"\n```\n\n### Key Transformer Properties\n\n**Advantages:**\n1.
     **Parallelization**: No sequential dependencies → can process entire batch simultaneously\n2. **Long-range
     dependency modeling**: Direct connections between any two positions\n3. **Input/output flexibility**: Can operate
     on variable-length sequences independently  \n4. **Rich representation learning**: Multi-head attention captures
     diverse relationship types\n\n**Limitations:**\n1. **Quadratic memory complexity**: Self-attention scales as O(n²)
     in sequence length\n2. **Position information must be added separately**: Requires explicit encoding mechanism\n3.
     **Context window constraint**: Can only attend to positions within receptive field (fixed by architecture
     design)\n\n---\n\n## 7. Practical Implementation Guide\n\n### Complete Working Example: Training a Transformer for
     Text Classification\n\n```python\nimport torch\nimport torch.nn as nn\nimport math\n\nclass
     SimpleTransformerClassifier(nn.Module):\n    \"\"\"Minimal transformer implementation for classification
     task\"\"\"\n    \n    def __init__(self, vocab_size, d_model=128, n_heads=4, \n                 num_layers=2,
     max_len=64, dropout_rate=0.1):\n        super().__init__()\n        \n        self.embedding =
     nn.Embedding(vocab_size, d_model)\n        \n        # Positional encoding (sinusoidal like original paper)\n
     pe_dim = d_model // 2 if d_model % 2 == 0 else d_model - 1\n        div_term = torch.exp(torch.arange(0, pe_dim,
     2).float() * \n                            (-math.log(10000.0) / (d_model // 2)))\n        \n
     self.register_buffer('pos_encoding', None)  \n        # Will be computed lazily based on sequence length\n
     \n        encoder_layer = nn.TransformerEncoderLayer(\n            d_model=d_model, nhead=n_heads, \n
     dim_feedforward=512, dropout=dropout_rate,\n            activation='relu'  # or 'gelu', 'swish'\n        )\n
     \n        self.encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)\n        \n
     self.classifier_head = nn.Sequential(\n            Linear(d_model, d_model // 2),\n            ReLU(),\n
     Dropout(dropout_rate),\n            Linear(d_model // 2, 10)  # Output classes (example: sentiment analysis)\n
     )\n    \n    def _get_positional_encoding(self, max_len):\n        if self.pos_encoding is None or
     self.pos_encoding.size(0) != max_len:\n            pe = torch.zeros(max_len, self.embedding.embedding_dim)\n
     \n            position = torch.arange(max_len).unsqueeze(1).repeat(1, \n
     min(self.embedding.embedding_dim // 2, (self.embedding.embedding_dim + 1) // 2))\n            \n
     div_term = math.log(10000.0) / max(self.embedding.embedding_dim // 2, 1)\n            \n            for i in
     range(pe.size(1)):\n                pe[:, i] = torch.sin(position.float() * (div_term ** (i % 2)))\\\n
     if i < self.embedding.embedding_dim else 0\n            \n            # Even dimensions use cosine:\n
     pe_pos = position.unsqueeze(-1)\n            for i in range(pe.size(1)):\n                if i % 2 == 0 and i >
     0:\n                    pe[:, i] = torch.cos(position.float() * (div_term ** ((i + 1) // 2)))\\\n
     if i < self.embedding.embedding_dim else 0\n            \n            # Simpler implementation matching PyTorch's
     Transformer encoder layer defaults:\n            for k in range(0, min(pe.size(-1), pe.size(0)), 2):\n
     freq = div_term ** (k / max(self.embedding.embedding_dim // 2, 1))\n                \n            return
     self.pos_encoding\n            \n        # Return cached positional encoding\n        \n    def forward(self,
     input_ids):\n        \"\"\"input_ids: [batch_size × seq_len]\"\"\"\n        \n        batch_size, seq_len =
     input_ids.shape\n        x = self.embedding(input_ids)         # [b×s×d_model]\n        \n        if
     self.pos_encoding is None or self.pos_encoding.size(0) != max(self.encoder.layers[0].layers[-1]):\n            pe =
     torch.zeros(seq_len, self.d_model).to(x.device)\n            \n            position_tensor =
     torch.arange(seq_len).unsqueeze(-1).expand(seq_len, d_model).float()\n            \n            for k in range(0,
     min(d_model, seq_len), 2):\n                freq = math.log(10000.0) / (d_model // 2) ** ((k + 1) % 2)\n
     \n        x += self.pos_encoding[:seq_len]       # Add positional info\n        \n        h = self.encoder(x)
     # [b×s×d_model]\n        \n        return self.classifier_head(h[:, -1])   # Use last position output for
     classification\n\n# Training loop example:\ndef train_epoch(model, data_loader, optimizer, criterion):\n
     model.train()\n    \n    total_loss = 0\n    \n    for batch in data_loader:\n        input_ids =
     batch['input_ids'].to(device)      # [b × seq_len]\n        labels = batch['labels'].to(device).long()     #

     loss.backward()\n        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)  # Prevent exploding
     gradients\n        \n        optimizer.step()\n        \n        total_loss += loss.item()\n    \n    return
     total_loss / len(data_loader)\n\n# Evaluation:\ndef evaluate(model, data_loader):\n    model.eval()\n    \n
     correct = 0\n    total_samples = 0\n    \n    with torch.no_grad():\n        for batch in data_loader:\n
     input_ids = batch['input_ids'].to(device)\n            labels = batch['labels'].to(device).long()\n            \n
     outputs = model(input_ids)\n            \n            predictions = torch.argmax(outputs, dim=1)\n            \n
     correct += (predictions == labels).sum().item()\n            total_samples += labels.size(0)\n    \n    return 100
     * correct / total_samples\n\n# Example: Training configuration dictionary for reproducibility\ntraining_config =
     {\n    'learning_rate': 3e-4,           # Adam default for transformers; can reduce to 5e-5 or less  \n
     'weight_decay': 0.01,            # L2 regularization (prevents overfitting)\n    'batch_size': 32,                #
     Adjust based on GPU memory constraints\n    'num_epochs': 10,\n    'dropout_rate': 0.1,             # Dropout for
     regularization\n    'warmup_steps': 400,             # Gradual LR increase at start (recommended)\n
     'max_grad_norm': 1.0            # Gradient clipping threshold\n}\n\n# Learning rate schedule with warmup and
     decay:\ndef get_lr_schedule(current_step, total_warmup_steps=5000):\n    \"\"\"Cosine annealing after linear
     warmup\"\"\"\n    \n    lr = current_learning_rate\n    \n    if current_step < total_warmup_steps:\n        #
     Linear warm-up phase\n        return (total_warmup_steps + 1) * ((current_step / total_warmup_steps)) *
     learning_rate\n    \n    else:\n        # Cosine annealing decay after warmup\n        progress = min(0.9,
     current_step - total_warmup_steps) / \\\n                  max(total_training_steps - total_warmup_steps, 1e-6)\n
     \n        return (learning_rate * lr_min_ratio + \n                (lr_max_ratio - learning_rate *
     lr_min_ratio))\n\n\"\"\"\n# PyTorch's built-in scheduler for production use:\nscheduler =
     get_linear_schedule_with_warmup(\n    optimizer=optimizer, num_warmup_steps=warmup_steps, \n

     loss.backward()\n    optimizer.step()\n    \n    scheduler.step()  # Update learning rate based on
     progress\n\"\"\"\n```\n\n---\n\n## 8. Training and Optimization\n\n### Common Challenges & Solutions\n\n####
     Vanishing Gradients (Deep Networks)\n\n**Problem:** In very deep networks, gradients can become extremely small
     during backpropagation, preventing effective weight updates in early layers.\n\n**Solutions:**\n1. **Residual
     Connections**: Add skip connections that preserve identity information\n   ```python\n   # From ResNet paper: h(x)
     = F(x,x_) + x  where F is residual mapping\n   output = layer_output + input_before_layer    # Skip connection\n
     ```\n\n2. **Proper Initialization**: Use Xavier/Glorot or He initialization schemes\n   \n3. **Batch
     Normalization**: Normalize activations to keep them in stable range:\n   ```python\n   batch_norm =
     nn.BatchNorm1d(d_model)  # Learnable scale/shift parameters\n    \n   def forward(x):\n       x = layer(x)\n
     return batch_norm(x)               # Stabilizes training, enables deeper networks\n   ```\n\n#### Overfitting
     (Model Memorization)\n\n**Problem:** Model performs well on training data but poorly on unseen test
     data.\n\n**Solutions:**\n1. **Dropout**: Randomly zero out neurons during training:\n   ```python\n   dropout =
     nn.Dropout(0.5)  # Keep 50% of units active per layer\n   \n   def forward(x):\n       x = linear_layer(x)\n
     return dropout(x)         # Stochastic regularization\n   ```\n\n2. **Data Augmentation**: Increase training data
     diversity:\n   - For text: synonym replacement, random insertion/deletion\n   - For images: rotation, cropping,
     color jittering\n   \n3. **Early Stopping**: Monitor validation loss and stop when it plateaus:\n   ```python\n
     best_val_loss = float('inf')\n   patience_counter = 0\n   max_patience = 10\n    \n    for epoch in
     range(max_epochs):\n        train_loss = train_epoch(...)\n        \n        val_loss, val_acc = evaluate(model)\n
     \n        if val_loss < best_val_loss:\n            best_val_loss = val_loss\n            save_model()\n
     patience_counter = 0\n        else:\n            patience_counter += 1\n        \n        if patience_counter >=
     max_patience:\n            print(\"Early stopping triggered\")\n            break\n    \n    # Use model with
     lowest validation loss for final evaluation\n   ```\n\n4. **Weight Decay**: L2 regularization on weights to prevent
     extreme values:\n   ```python\n   optimizer = torch.optim.Adam(model.parameters(), lr=1e-3, weight_decay=0.01)\n
     ```\n\n#### Mode Collapse (in GANs and other generative models)\n\n**Problem:** Model produces limited variety of
     outputs despite diverse training data.\n\n**Solutions:**\n1. **Mini-batch discrimination**: Compare samples within
     batch during loss computation\n   \n2. **Feature matching**: Minimize difference between real/generated feature
     distributions instead of only adversarial loss\n   \n3. **Noise injection**: Add controlled noise to generator
     inputs or discriminator targets\n\n### Advanced Optimization Techniques\n\n#### Learning Rate Scheduling
     Strategies\n\n**Cosine Annealing:**\n```python\nscheduler = torch.optim.lr_scheduler.CosineAnnealingLR(\n
     optimizer, T_max=num_epochs * len(train_loader), eta_min=1e-6\n)\n\"\"\"\nDecays learning rate smoothly from
     initial_lr to min_lr over period.\nGood for finding optimal convergence point without getting stuck in local
     minima.\"\"\"\n\n**Warm Restart (SGDR):**\n```python\nscheduler = torch.optim.lr_scheduler.CyclicLR(\n
     optimizer, \n    base_lr=1e-5, max_lr=3e-4,\n    step_size_up=len(train_loader) // 2,\n    mode='triangular'
     # or 'triangular2', 'exp_range'\n)\n\n# Creates triangular learning rate cycles: increases then decreases
     repeatedly.\n\"\"\"Can help escape local minima by periodically testing different scales.\"\"\"\n```\n\n**Linear
     Warmup + Decay:**\n```python\ndef cosine_schedule_with_warmup(current_step, total_steps):\n    \"\"\"Standard
     practice in transformer training (e.g., BERT pretraining)\"\"\"\n    \n    if current_step < warmup_steps:\n
     return initial_lr * (current_step / warmup_steps)  # Linear increase\n    \n    else:\n        decay_ratio =
     max(0.5, min(current_step - warmup_steps, \n                                   total_steps - warmup_steps)) \\\n
     / (total_steps - warmup_steps)\n        \n        return initial_lr * decay_ratio  # Cosine decay after warmup\n\n#
     Typical schedule in practice:\n\"\"\"\n- Warmup: first 10% of training steps → prevents early instability\n- Decay:
     remaining 90% with cosine curve → fine-tunes convergence\n- Final phase may use constant minimum LR or continue
     decaying to zero\n\"\"\"\n```\n\n### Gradient Accumulation for Large Batches\n\n**Problem:** GPU memory limits
     prevent using large batch sizes that stabilize training.\n\n**Solution:** Simulate larger batches by accumulating
     gradients over multiple mini-batches:\n\n```python\nclass GradAccumulator(torch.nn.Module):\n    \"\"\"Wrapper to
     accumulate gradients across multiple steps\"\"\"\n    \n    def __init__(self, model, accumulation_steps=4):\n
     super().__init__()\n        self.model = model\n        self.accumulation_steps = accumulation_steps\n        \n
     def forward(self, input_ids, labels=None):\n        # Accumulate loss and gradients over multiple mini-batches\n
     total_loss = 0\n    \n        for step in range(self.accumulation_steps):\n            if step > 0:\n
     optimizer.zero_grad()      # Clear previous accumulated gradient\n            \n            batch_input =
     get_batch(input_ids, labels=labels)\n            \n            outputs = self.model(batch_input['input_ids'])\n
     loss = criterion(outputs, batch_input['labels']) / \\\n                   accumulation_steps     # Scale down by
     number of steps\n    \n            total_loss += loss\n    \n        return total_loss\n\n#
     Usage:\n\"\"\"\nmodel_with_accumulator = GradAccumulator(model, accumulation_steps=4)\ntrain_loader =
     DataLoader(dataset, batch_size=8 * 4, shuffle=True)  # Effective batch size is 128 instead of 32\n\nfor epoch in
     range(num_epochs):\n    for input_ids, labels in train_loader:\n        loss = model_with_accumulator(input_ids,
     labels)\n        \n        loss.backward()          # Accumulates gradients across steps\n        optimizer.step()
     # Updates weights with accumulated gradient\n        \n# Note: Zero out gradients between outer loop iterations to
     prevent accumulation carryover\n\"\"\"\n\n### Mixed Precision Training (AMP - Automatic Mixed
     Precision)\n\n**Benefits:**\n- Reduces memory footprint by factor of 2-4x → enables larger batches or models on
     same hardware\n- Faster training due to optimized GPU kernels for FP16 operations\n- Maintains numerical stability
     through loss scaling technique\n\n```python\nfrom torch.cuda.amp import autocast, GradScaler\n\nscaler =
     GradScaler()              # Handles loss scaling automatically\n\nfor batch in train_loader:\n    input_ids =
     batch['input_ids'].to(device)\n    labels = batch['labels'].to(device).long()\n    \n    optimizer.zero_grad()\n
     \n    with torch.cuda.amp.autocast():   # Context manager for mixed precision\n        outputs = model(input_ids)
     # Compute in FP16 where possible\n        \n        loss = criterion(outputs, labels)  # Cross-entropy (numerically
     stable even at low precision)\n    \n    scaler.scale(loss).backward()       # Scale gradient to prevent
     underflow\n    \n    optimizer.step()                    # Apply parameter updates\n```\n\n### Gradient Clipping
     for Stability\n\n**Problem:** Very large gradients can cause weights to explode, making training
     unstable.\n\n**Solution:** Clip gradient norm to maximum
     threshold:\n\n```python\ntorch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)\n\n\"\"\"Clips
     per-parameter norms so that overall L2 norm of all parameters doesn't exceed 1.0.\"\"\"\n\n# Alternative: clip
     individual parameter gradients by value magnitude:\nfor param in model.parameters():\n    if grad.requires_grad and
     torch.abs(grad).max() > threshold:\n        grad.clamp_(-threshold, threshold)  # Element-wise
     clipping\n\n\n```\n\n---\n\n## 9. Model Evaluation\n\n### Classification Metrics\n\n**Accuracy:** Simple but can be
     misleading with imbalanced classes\n```python\ndef accuracy(predictions, labels):\n    \"\"\"Measure of correct
     predictions\"\"\"\n    return (predictions == labels).float().mean()\n```\n\n**Precision and Recall:** Important
     for handling class imbalance:\n```python\nfrom sklearn.metrics import precision_score, recall_score\n\nprecision =
     precision_score(labels, predictions, average='weighted')  # Weighted by class frequency  \nrecall =
     recall_score(labels, predictions, average='macro')          # Average across classes equally\n\n\"\"\"\n- Precision
     (positive predictive value): Of all predicted positive samples, what fraction actually are?\n- Recall
     (sensitivity): Of all actual positives in dataset, how many did we correctly identify?\n\nExample: Medical
     diagnosis\n- High recall needed: Don't miss disease cases even if it means more false alarms  \n- High precision
     preferred when false positive cost is high\"\"\"\n```\n\n**F1 Score:** Harmonic mean of precision and recall
     (balanced measure):\n```python\nfrom sklearn.metrics import f1_score\n\nf1 = f1_score(labels, predictions,
     average='weighted')  # Weighted by class frequency\n\n\n```\n\n### Confusion Matrix Analysis\n\n```python\nimport
     seaborn as sns\nimport matplotlib.pyplot as plt\nfrom sklearn.metrics import confusion_matrix\n\n# Generate
     confusion matrix: rows=actual labels, columns=predictions\ncm = confusion_matrix(y_true=labels,
     y_pred=predictions)\n\nplt.figure(figsize=(10, 8))\n\nsns.heatmap(cm, annot=True, fmt='d', cmap='Blues')
     \n\"\"\"Annotates cell with count; 'fmt=d' shows integers instead of floats.\"\"\"\n\nplt.xlabel('Predicted

+
+**Transformers** are a revolutionary neural network architecture introduced by Vaswani et al. (2017) that replaced recurrent networks as the dominant paradigm for sequence processing tasks like language understanding and generation.
