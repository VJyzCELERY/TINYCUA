\begin{algorithm}[t]
\caption{TINYCUA Main Workflow}
\label{alg:tinycua-main-workflow}
\begin{algorithmic}[1]
\Require User query $q$, session context $C$, worker effort $e$
\Ensure Final user response $r$

\State \textbf{// Phase 1: Query Analyst}
\State $q_{\text{enhanced}},\; \textit{route} \gets \textsc{QueryAnalystNode}(q,\; C)$

\If{$\textit{route} = \textsc{Passthrough}$}
    \State \Return $\textsc{ResponseNode}(q_{\text{enhanced}},\; C)$
\EndIf

\State \textbf{// Phase 2: Information Digester}
\State $d \gets \textsc{InformationDigesterNode}(q_{\text{enhanced}},\; C)$

\State \textbf{// Phase 3: Worker Routing}
\State $\textit{workerRoute} \gets \textsc{WorkerNode}(d)$

\If{$\textit{workerRoute} = \textsc{Passthrough}$}
    \State \Return $\textsc{ResponseNode}(d,\; C)$
\EndIf

\State \textbf{// Phase 4: Task Tree Creation}
\If{$\textit{workerRoute} = \textsc{TaskCreation}$}
    \State $T \gets \textsc{TaskCreateNode}(d,\; e)$
\ElsIf{$\textit{workerRoute} \in \{\textsc{TaskRecreation},\; \textsc{TaskReanalysis}\}$}
    \State $T \gets \textsc{TaskAnalyzerNode}(d)$
\EndIf

\State \textbf{// Phase 5: Analysis-Effort Loop}
\State $\textit{passCount} \gets 0$
\State $\textit{passLimit} \gets \textsc{AnalysisEffortNode}(e)$

\While{$\textit{passCount} < \textit{passLimit}$}
    \State $\textit{selectedTasks} \gets \textsc{TaskAssessorNode}(T)$
    \If{$\textit{selectedTasks} = \emptyset$}
        \State \textbf{break}
    \EndIf
    \ForAll{$\textit{task} \in \textit{selectedTasks}$}
        \State $\textit{children} \gets \textsc{TaskAnalyzerNode}(\textit{task.Context})$
        \State $\textsc{AttachChildren}(T,\; \textit{task},\; \textit{children})$
    \EndFor
    \State $\textit{passCount} \gets \textit{passCount} + 1$
\EndWhile

\State \textbf{// Phase 6: Sequential Task Execution Loop}
\While{$\exists$ unfinished leaf task in $T$}
    \State $\textit{task} \gets \textsc{SelectNextUnfinishedLeaf}(T)$
    \State $\textit{result} \gets \textsc{TaskExecutorNode}(\textit{task},\; \textsc{ShallowRoadmap}(T))$

    \State $\textit{review} \gets \textsc{ResultReviewerNode}($
    \quad $\textit{task},\; \textit{result},\; \textsc{ExecutionLog}(\textit{task}),\; \textsc{ShallowRoadmap}(T)$
    \ $)$

    \If{$\textit{review.Status} = \textsc{Accept}$}
        \If{$\textit{review.RootDone}$}
            \State $\textsc{PropagateContext}(T,\; \textit{task},\; \textit{review})$
            \State $\textsc{MarkAccepted}(T,\; \textit{task},\; \textit{result},\; \textit{review})$
            \State \textbf{break}
        \Else
            \State $\textsc{PropagateContext}(T,\; \textit{task},\; \textit{review})$
            \State $\textsc{MarkAccepted}(T,\; \textit{task},\; \textit{result},\; \textit{review})$
        \EndIf

    \ElsIf{$\textit{review.Status} = \textsc{Retry}$}
        \State $\textsc{RecordFailureContext}(\textit{task},\; \textit{review})$

    \ElsIf{$\textit{review.Status} = \textsc{Replan}$}
        \State $\textit{subtasks} \gets \textsc{TaskAnalyzerNode}(\textit{task.Context},\; \textit{review.Rationale})$
        \State $\textsc{ReplaceLeafWithSubtasks}(T,\; \textit{task},\; \textit{subtasks})$

        \State \textbf{// Re-enter decomposition loop}
        \State $\textit{passCount} \gets 0$
        \State $\textit{passLimit} \gets \textsc{AnalysisEffortNode}(e)$
        \While{$\textit{passCount} < \textit{passLimit}$}
            \State $\textit{selectedTasks} \gets \textsc{TaskAssessorNode}(T)$
            \If{$\textit{selectedTasks} = \emptyset$}
                \State \textbf{break}
            \EndIf
            \ForAll{$\textit{selectedTask} \in \textit{selectedTasks}$}
                \State $\textit{children} \gets \textsc{TaskAnalyzerNode}(\textit{selectedTask.Context})$
                \State $\textsc{AttachChildren}(T,\; \textit{selectedTask},\; \textit{children})$
            \EndFor
            \State $\textit{passCount} \gets \textit{passCount} + 1$
        \EndWhile
    \EndIf
\EndWhile

\State \textbf{// Phase 7: Result Aggregation and Response}
\State $a \gets \textsc{ResultAggregationNode}(T)$
\State \Return $\textsc{ResponseNode}(a)$
\end{algorithmic}
\end{algorithm}
