\begin{algorithm}[t]
\caption{TINYCUA Query Analyst Node}
\label{alg:query-analyst}
\begin{algorithmic}[1]
\Require User query $q$, session context $C$
\Ensure Enhanced query $q_{\text{enhanced}}$, route label $\mathit{route}$

// Deduplication guard
\If{\textsc{QueryAnalystAlreadyActive}($C$)}
    \State \Return $\bot$
\EndIf

// Context collection (read-only)
\State $\mathit{ctx} \gets \textsc{CollectContext}(q,\; C)$

// Build reasoning window and classify
\State $\mathit{rw} \gets \{q\} \cup \textsc{FilterRelevant}(\mathit{ctx},\; q)$
\State $\mathit{route} \gets \textsc{ClassifyRoute}(\textsc{SLMAnalyze}(\mathit{rw}))$

// Validate route with retry
\While{$\mathit{route}$ is invalid \textbf{or} not invoked}
    \State $\mathit{route} \gets \textsc{ClassifyRoute}(\textsc{SLMAnalyze}(\mathit{rw}))$
    \If{retries exceeded}
        \State \Return $q,\; \textsc{DefaultRoute}()$
    \EndIf
\EndWhile

// Dispatch
\State $q_{\text{enhanced}} \gets \{q\} \cup \textsc{FilterRelevant}(\mathit{rw},\; q)$
\If{$\mathit{route} = \textsc{Worker}$}
    \State $\textsc{SpawnInformationDigester}()$
\EndIf
\State \Return $q_{\text{enhanced}},\; \mathit{route}$
\end{algorithmic}
\end{algorithm}
