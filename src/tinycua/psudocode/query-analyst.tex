\begin{algorithm}[t]
\caption{TINYCUA Query Analyst Node}
\label{alg:query-analyst}
\begin{algorithmic}[1]
\Require User query $q$, session context $C$
\Ensure Enhanced query $q_{\text{enhanced}}$, route label $\mathit{route}$

\Statex \textit{--- Deduplication guard ---}
\If{\textsc{QueryAnalystAlreadyActive}($C$)}
    \State \Return $\bot$
\EndIf

\Statex \textit{--- Context collection (read-only) ---}
\State $\mathit{rootCtx} \gets \textsc{ReadRootSession}(C)$
\State $\mathit{nodeCtx} \gets \textsc{InspectExistingNodes}(C)$
\State $\mathit{exploratoryCtx} \gets \textsc{ExploreInformation}(q,\; C)$

\Statex \textit{--- Temporary reasoning window (not merged back to parent) ---}
\State $\mathit{reasoningWindow} \gets \big\{$
\quad $q,\;$
\quad $\textsc{FilterRelevant}(\mathit{rootCtx},\; q),\;$
\quad $\textsc{FilterRelevant}(\mathit{nodeCtx},\; q),\;$
\quad $\mathit{exploratoryCtx}$
\ $\big\}$

\Statex \textit{--- SLM classification ---}
\State $\mathit{analysis} \gets \textsc{SLMAnalyze}(\mathit{reasoningWindow})$
\State $\mathit{route} \gets \textsc{ClassifyRoute}(\mathit{analysis})$

\Statex \textit{--- Route validation and retry ---}
\State $\mathit{retryCount} \gets 0$
\While{$\mathit{route}$ is invalid \textbf{or} $\mathit{route}$ is not invoked}
    \State $\mathit{retryCount} \gets \mathit{retryCount} + 1$
    \If{$\mathit{retryCount} > \textsc{NodeRetryPolicy.MaxRetries}$}
        \State $\mathit{route} \gets \textsc{DefaultRoute}()$
        \State \textbf{break}
    \EndIf
    \State $\mathit{analysis} \gets \textsc{SLMAnalyze}(\mathit{reasoningWindow})$
    \State $\mathit{route} \gets \textsc{ClassifyRoute}(\mathit{analysis})$
\EndWhile

\Statex \textit{--- Route dispatch ---}
\State $q_{\text{enhanced}} \gets \big\{$
\quad $q,\;$
\quad $\textsc{FilterRelevant}(\mathit{reasoningWindow},\; q)$
\ $\big\}$

\If{$\mathit{route} = \textsc{Worker}$}
    \State $\textsc{SpawnInformationDigester}()$
    \State \Return $q_{\text{enhanced}},\; \textsc{Worker}$
\ElsIf{$\mathit{route} = \textsc{Passthrough}$}
    \State \Return $q_{\text{enhanced}},\; \textsc{Passthrough}$
\EndIf
\end{algorithmic}
\end{algorithm}
