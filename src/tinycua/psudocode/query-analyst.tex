\begin{algorithm}[t]
\caption{TINYCUA Query Analyst}
\label{alg:query-analyst}
\begin{algorithmic}[1]
\Require $\mathit{root\_session}$, $\mathit{user\_query}$, $\mathit{queue}$
\Ensure $\mathit{route} \in \{\textsc{Worker}, \textsc{Passthrough}, \textsc{Uncertain}\}$

// Step 1: Deduplication guard
\If{$\mathit{queue}$ contains active $\mathit{QueryAnalyst}$}
    \State \Return mandatory\_passthrough($\mathit{QueryAnalyst}$)
\EndIf

// Step 2: Deterministic precheck
\If{$\exists$ valid mandatory\_passthrough}
    \State forward continuation to target node/session
    \State \Return
\EndIf

// Step 3: Assemble transient reasoning window
\State $\mathit{window} \gets \mathit{root\_session}.\mathit{context} \;\Vert\; \mathit{queue}.\mathit{contexts} \;\Vert\; \mathit{user\_query}$

// Step 4: Two-step classification (LLM + tool)
\State $\mathit{route} \gets \bot$; $\mathit{valid} \gets \textsc{False}$
\While{$\neg\mathit{valid}$}
    \State $\mathit{analysis} \gets \textsc{SLMAnalyze}(\mathit{window})$
    \State $\mathit{route} \gets \textsc{Classify}(\mathit{analysis})$; $\mathit{valid} \gets \mathit{route} \in \{\textsc{Worker}, \textsc{Passthrough}, \textsc{Uncertain}\}$
\EndWhile

// Step 5: Dispatch via route map
\State \textsc{Dispatch}($\mathit{route}$)
\If{$\mathit{route} = \textsc{Worker}$}
    \State spawn \textsc{InformationDigester} before \textsc{WorkerNode}
\EndIf

\end{algorithmic}
\end{algorithm}
