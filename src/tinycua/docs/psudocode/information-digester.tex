\begin{algorithm}[t]
\caption{TINYCUA Information Digester Node}
\label{alg:information-digester}
\begin{algorithmic}[1]
\Require Session context $C$, worker query $q$
\Ensure Context cache path $\mathit{cachePath}$, digested summary $\mathit{digest}$

// Step 1: Collect session contexts
\State $\mathit{rootCtx} \gets \textsc{ReadRootSession}(C)$
\State $\mathit{agentSessions} \gets \textsc{ListActiveAgentSessions}(C)$

// Step 2: Assemble unified context cache
\State $\mathit{allContexts} \gets [\mathit{rootCtx}]$
\ForAll{$s \in \mathit{agentSessions}$}
    \State $\mathit{agentCtx} \gets \textsc{ReadAgentSession}(s)$
    \State $\mathit{allContexts}.\textsc{Append}(\mathit{agentCtx})$
\EndFor
\State $\mathit{cachePath} \gets \textsc{WriteCacheFile}(\mathit{allContexts})$

// Step 3: Spawn enhanced context retrieval (ReAct agent)
\State $\mathit{retrievalAgent} \gets \textsc{SpawnReActAgent}(\mathit{cachePath})$

// Step 4: Grep-based context retrieval
\State $\mathit{retrievedCtx} \gets \textsc{ReActRetrieve}(\mathit{retrievalAgent},\; q)$

// Step 5: Invoke digest information tool
\State $\mathit{digestTool} \gets \textsc{InitDigestTool}(\mathit{cachePath})$
\State $\mathit{digest} \gets \textsc{DigestInformation}(\mathit{digestTool},\; q,\; \mathit{retrievedCtx})$

// Step 6: Ensure at least one digest operation
\If{$\mathit{digest} = \emptyset$}
    \State $\mathit{digest} \gets \textsc{DigestInformation}(\mathit{digestTool},\; q,\; \mathit{retrievedCtx})$
\EndIf

\end{algorithmic}
\end{algorithm}
