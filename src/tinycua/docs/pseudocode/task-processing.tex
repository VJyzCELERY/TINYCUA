\begin{algorithm}[t]
\caption{TINYCUA Task Processing}
\label{alg:task-processing}
\begin{algorithmic}[1]
\Require $\mathit{route}$, $\mathit{queue}$, $\mathit{effort}$
\Ensure $\mathit{aggregatedResult}$

\Statex \textcolor{gray}{// Step 1: Task Creation}
\If{$\mathit{route} = \text{task\_creation}$}
    \State $\mathit{rootTask}, \mathit{summary} \gets \textsc{TaskCreate}(\mathit{query})$
\EndIf

\Statex \textcolor{gray}{// Step 2: Initial Decomposition}
\State $\textsc{TaskAnalyze}(\mathit{rootTask}, \text{initial\_analysis})$

\Statex \textcolor{gray}{// Step 3: Effort-Gated Analysis Loop}
\State $\mathit{pass} \gets 0$
\While{$\textsc{AnalysisEffort}(\mathit{effort}, \mathit{pass}) = \text{continue}$}
    \State $\mathit{selected} \gets \textsc{TaskAssess}(\mathit{rootTask})$; \ \If{$\mathit{selected} = \textsc{null}$} \textbf{break} \EndIf
    \State $\textsc{TaskAnalyze}(\mathit{selected}, \text{effort\_loop})$; \ $\mathit{pass} \gets \mathit{pass} + 1$
\EndWhile

\Statex \textcolor{gray}{// Step 4: Execute and Review Loop}
\State $\mathit{completedResults} \gets \textsc{null}$
\State $\mathit{active} \gets \textsc{NextTask}(\mathit{rootTask})$
\While{$\mathit{active} \neq \textsc{null}$}
    \State $\mathit{context} \gets \textsc{BuildContext}(\mathit{active}, \mathit{completedResults})$
    \State $\mathit{result} \gets \textsc{TaskExecute}(\mathit{active}, \mathit{context})$; \ $\mathit{retries} \gets 0$
    \State $\mathit{decision} \gets \textsc{ResultReview}(\mathit{result})$
    \While{$\mathit{decision} = \text{reject} \ \wedge \ \mathit{retries} < \mathit{retryLimit}$}
        \State $\mathit{result} \gets \textsc{TaskExecute}(\mathit{active}, \mathit{context})$; \ $\mathit{retries} \gets \mathit{retries} + 1$; \ $\mathit{decision} \gets \textsc{ResultReview}(\mathit{result})$
    \EndWhile
    \If{$\mathit{decision} = \text{replan}$}
        \State $\textsc{TaskAssess}(\mathit{active})$; \ $\textsc{TaskAnalyze}(\mathit{active}, \text{replan})$; \ $\mathit{result} \gets \textsc{TaskExecute}(\mathit{active}, \mathit{context})$
    \EndIf
    \If{$\mathit{decision} = \text{accept}$}
        \State $\mathit{completedResults} \gets \mathit{completedResults} \cup \{\mathit{result}\}$; \ $\mathit{active} \gets \textsc{NextTask}(\mathit{rootTask})$
    \EndIf
\EndWhile

\Statex \textcolor{gray}{// Step 5: Aggregate Results}
\State $\mathit{aggregatedResult} \gets \textsc{Aggregate}(\mathit{rootTask})$

\end{algorithmic}
\end{algorithm}
